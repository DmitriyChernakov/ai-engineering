Файл `week01/day04_structured.py`:

```python
import ollama
import json
from pydantic import BaseModel, field_validator

class ReviewAnalysis(BaseModel):
    sentiment: str
    pros: list[str]
    cons: list[str]
    confidence: float

    @field_validator('sentiment')
    def validate_sentiment(cls, v):
        allowed = ['positive', 'negative', 'mixed']
        if v not in allowed:
            raise ValueError(f'Invalid sentiment: {v}. Must be one of {allowed}')
        return v

    @field_validator('confidence')
    def validate_confidence(cls, v):
        if not 0.0 <= v <= 1.0:
            raise ValueError('Confidence must be between 0 and 1')
        return v

review = "Доставка была быстрой, но упаковка помята. Сам товар работает отлично."

prompt = f"""Проанализируй отзыв и верни результат строго в JSON со следующей структурой:
{{"sentiment": "positive|negative|mixed", "pros": [...], "cons": [...], "confidence": 0.0-1.0}}
Отзыв: "{review}"
"""

response = ollama.chat(
    model='qwen2.5:7b',
    messages=[{'role': 'user', 'content': prompt}],
    format='json'  # Ключевой параметр: заставляет Ollama возвращать валидный JSON
)

raw = response['message']['content']
parsed = ReviewAnalysis.model_validate_json(raw)

print(f"Тональность: {parsed.sentiment}")
print(f"Плюсы: {parsed.pros}")
print(f"Минусы: {parsed.cons}")
print(f"Уверенность: {parsed.confidence}")
```

Файл `week01/qa_buddy.py`:

```python
import ollama
import json
import sys

SYSTEM_PROMPT = """Ты — опытный QA-инженер с 10-летним стажем в автоматизации.
Твоя задача: получить описание фичи и сгенерировать тест-кейсы.

Правила:
1. Генерируй от 3 до 7 тест-кейсов
2. Включай позитивные, негативные и граничные сценарии
3. Для каждого кейса укажи: название, шаги, ожидаемый результат, приоритет
4. Отвечай строго в JSON

Формат:
{
  "test_cases": [
    {
      "id": 1,
      "name": "Краткое название",
      "type": "positive|negative|edge_case",
      "steps": ["шаг 1", "шаг 2"],
      "expected_result": "Что должно произойти",
      "priority": "high|medium|low"
    }
  ]
}"""

def generate_test_cases(feature_description: str) -> dict:
    """Генерирует тест-кейсы из описания фичи."""
    response = ollama.chat(
        model='qwen2.5:7b',
        messages=[
            {'role': 'system', 'content': SYSTEM_PROMPT},
            {'role': 'user', 'content': f'Сгенерируй тест-кейсы для фичи: {feature_description}'}
        ],
        format='json'
    )
    return json.loads(response['message']['content'])

def save_to_file(result: dict, filename: str) -> None:
    """Сохраняет результат в JSON-файл."""
    with open(filename, 'w', encoding='utf-8') as f:
        json.dump(result, f, indent=2, ensure_ascii=False)
    print(f"Сохранено в {filename}")

if __name__ == '__main__':
    if len(sys.argv) > 1:
        feature = ' '.join(sys.argv[1:])
    else:
        feature = input('Опиши фичу: ')

    print('\nГенерирую тест-кейсы...\n')
    try:
        result = generate_test_cases(feature)
        print(json.dumps(result, indent=2, ensure_ascii=False))
        save_to_file(result, 'test_cases_output.json')
    except Exception as e:
        print(f"Ошибка: {e}")
```

```python
import ollama
import json

def search_web(query: str) -> str:
    """Заглушка поиска в интернете."""
    return f"Результаты поиска по запросу '{query}': Python 3.12 вышел в октябре 2023..."

def calculate(expression: str) -> str:
    """Вычисляет математическое выражение."""
    try:
        return str(eval(expression))
    except Exception:
        return "Ошибка вычисления"

TOOLS = {
    "search_web": search_web,
    "calculate": calculate
}

SYSTEM = """У тебя есть инструменты:
1. search_web(query) — поиск информации
2. calculate(expression) — калькулятор

Если нужен инструмент, ответь строго в JSON:
{"tool": "имя_инструмента", "args": {"параметр": "значение"}}

Если инструмент не нужен, просто ответь текстом."""

def run_agent(user_message: str) -> None:
    messages = [
        {"role": "system", "content": SYSTEM},
        {"role": "user", "content": user_message}
    ]
    response = ollama.chat(model="qwen2.5:7b", messages=messages, format="json")
    content = response["message"]["content"]

    try:
        action = json.loads(content)
        if "tool" in action:
            tool_name = action["tool"]
            tool_args = action.get("args", {})
            print(f"[Агент выбрал инструмент: {tool_name}({tool_args})]")
            result = TOOLS[tool_name](**tool_args)
            print(f"[Результат: {result}]")

            # Второй проход: отдаём результат обратно модели
            messages.append({"role": "assistant", "content": content})
            messages.append({"role": "user", "content": f"Результат: {result}. Ответь пользователю."})
            final = ollama.chat(model="qwen2.5:7b", messages=messages)
            print(f"\nОтвет: {final['message']['content']}")
        else:
            print(f"\nОтвет: {content}")
    except json.JSONDecodeError:
        print(f"\nОтвет: {content}")

# Тесты
run_agent("Сколько будет 1024 * 768?")
run_agent("Найди информацию про Python 3.12")
run_agent("Привет, как дела?")
```

def multiply(a: int, b: int) -> int:

    return a * b

@tool
def get_weather(city: str) -> str:
    """Возвращает погоду в городе (заглушка)."""

**Цель дня:** Агент останавливается и спрашивает подтверждение перед «опасным» действием.

Файл `week02/day11_hitl.py`:

```python
from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import ToolNode, tools_condition
from langgraph.checkpoint.memory import MemorySaver
from langchain_ollama import ChatOllama
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage
from typing import TypedDict, Annotated
import operator

@tool
def send_email(to: str, subject: str, body: str) -> str:
    """Отправляет email. ОПАСНОЕ ДЕЙСТВИЕ — требует подтверждения."""
    return f"Email отправлен на {to}: '{subject}'"

@tool
def search_contacts(name: str) -> str:
    """Ищет контакт по имени."""
    contacts = {"Иван": "ivan@example.com", "Мария": "maria@example.com"}
    return contacts.get(name, "Контакт не найден")

tools = [send_email, search_contacts]
llm = ChatOllama(model="qwen2.5:7b").bind_tools(tools)
memory = MemorySaver()

class State(TypedDict):
    messages: Annotated[list, operator.add]

def chatbot(state: State):
    return {"messages": [llm.invoke(state["messages"])]}

graph = StateGraph(State)
graph.add_node("chatbot", chatbot)
graph.add_node("tools", ToolNode(tools=tools))
graph.add_edge(START, "chatbot")
graph.add_conditional_edges("chatbot", tools_condition)
graph.add_edge("tools", "chatbot")

# Ключевое: остановка ПЕРЕД вызовом инструментов
app = graph.compile(checkpointer=memory, interrupt_before=["tools"])
config = {"configurable": {"thread_id": "1"}}

# Шаг 1: запрос пользователя
result = app.invoke(
    {"messages": [HumanMessage(content="Отправь email Ивану с темой 'Привет'")]},
    config
)
print("Агент хочет выполнить действие.")
print("Подтвердить? (y/n)")

# Шаг 2: человек принимает решение
if input().lower() == "y":
    result = app.invoke(None, config)  # Продолжить выполнение
    print(result["messages"][-1].content)
else:
    print("Действие отменено.")
```

- Заметка `01-Concepts/Human-in-the-Loop.md`

```python
import os
import json

os.environ["LANGFUSE_PUBLIC_KEY"] = "pk-..."
os.environ["LANGFUSE_SECRET_KEY"] = "sk-..."
os.environ["LANGFUSE_HOST"] = "https://cloud.langfuse.com"

from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import ToolNode, tools_condition
from langchain_ollama import ChatOllama
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage, SystemMessage
from langfuse.callback import CallbackHandler
from typing import TypedDict, Annotated
import operator

@tool
def read_file(filepath: str) -> str:
    """Читает содержимое файла."""
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        return f"Файл '{filepath}' не найден"

@tool
def save_test_cases(filename: str, content: str) -> str:
    """Сохраняет тест-кейсы в файл."""
    with open(filename, 'w', encoding='utf-8') as f:
        f.write(content)
    return f"Тест-кейсы сохранены в '{filename}'"

@tool
def count_test_cases(test_cases_json: str) -> str:
    """Считает статистику по тест-кейсам."""
    try:
        data = json.loads(test_cases_json)
        cases = data.get("test_cases", [])
        stats = {}
        for case in cases:
            t = case.get("type", "unknown")
            stats[t] = stats.get(t, 0) + 1
        return json.dumps(stats, ensure_ascii=False)
    except json.JSONDecodeError:
        return "Ошибка парсинга JSON"

SYSTEM_PROMPT = """Ты — AI QA-инженер. Генерируй тест-кейсы по описанию фичи.

У тебя есть инструменты:
- read_file: прочитать описание фичи из файла
- save_test_cases: сохранить результат в файл
- count_test_cases: посчитать статистику

Процесс:
1. Если пользователь даёт путь к файлу — прочитай его через read_file
2. Сгенерируй тест-кейсы (positive, negative, edge_case)
3. Сохрани результат через save_test_cases
4. Покажи статистику через count_test_cases

Формат тест-кейсов:
{"test_cases": [{"id": 1, "name": "...", "type": "positive|negative|edge_case", "steps": [...], "expected_result": "...", "priority": "high|medium|low"}]}"""

tools = [read_file, save_test_cases, count_test_cases]
llm = ChatOllama(model="qwen2.5:7b").bind_tools(tools)

class State(TypedDict):
    messages: Annotated[list, operator.add]

def agent(state: State):
    return {"messages": [llm.invoke(state["messages"])]}

graph = StateGraph(State)
graph.add_node("agent", agent)
graph.add_node("tools", ToolNode(tools=tools))
graph.add_edge(START, "agent")
graph.add_conditional_edges("agent", tools_condition)
graph.add_edge("tools", "agent")
app = graph.compile()

# Запуск с Langfuse
langfuse = CallbackHandler()
result = app.invoke(
    {"messages": [
        SystemMessage(content=SYSTEM_PROMPT),
        HumanMessage(content="Сгенерируй тест-кейсы для фичи: авторизация через Google OAuth. Сохрани в output.json")
    ]},
    config={"callbacks": [langfuse]}
)
print(result["messages"][-1].content)
```
